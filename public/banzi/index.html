	if (state.size() == 1) return 1;

		bool ok = 1;

			if (!nxt[ret].empty()) {
				if (!dfs(nxt[ret], depth - 1, opt)) {
					ok = 0;
					break;
				}
			}
		}
		if (ok) {
			if (!solution.count(state)) solution[state] = query;
			return 1;
		}
	}
	return 0;
}
